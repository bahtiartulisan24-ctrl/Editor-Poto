                    return;
                }
            }

            // Cek Login Siswa
            const student = appState.students.find(s => s.username.toLowerCase() === username.toLowerCase());
            if (student) {
                if (student.password === password) {
                    currentUser = { role: 'student', data: student };

                    navigateTo('student-lobby');
                    return;

                    showModal('Password Salah', 'Password untuk akun siswa tersebut tidak sesuai.');
                    return;
                }
            }

            showModal('Akun Tidak Ditemukan', `Username <strong>${username}</strong> tidak terdaftar di database SD Negeri Seganteng.`);
        }

            event.preventDefault();
            const editId = document.getElementById('editQuestionId').value;
            const qObj = {
                id: editId || 'q_' + Date.now(),
                subject: document.getElementById('qSubject').value.trim(),

        function handleExcelQuestionUpload(event) {
            const file = event.target.files[0];
            if (!file) return;
            const defaultSubject = document.getElementById('importDefaultSubject').value;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    let rows = [];
                    if (window.XLSX) {
                        const data = new Uint8Array(e.target.result);
                        const workbook = XLSX.read(data, { type: 'array' });
                        const firstSheet = workbook.Sheets[workbook.SheetNames[0]];
                        rows = XLSX.utils.sheet_to_json(firstSheet, { defval: '' });
                    } else {
                        // Fallback CSV Parser
                        const text = new TextDecoder().decode(e.target.result);
                        rows = parseSimpleCSV(text);
                    }

                    const parsedQuestions = rows.map(row => {
                        // Normalisasi nama kolom (case-insensitive)
                        const getCol = (keys) => {
                            for (const k of Object.keys(row)) {
                                if (keys.some(target => k.trim().toLowerCase().includes(target))) {
                                    return String(row[k]).trim();
                                }
                            }
                            return '';
                        };

                        const subj = getCol(['mapel', 'mata pelajaran']) || defaultSubject;
                        const text = getCol(['pertanyaan', 'soal']);
                        const optA = getCol(['opsi a', 'pilihan a', 'a']);
                        const optB = getCol(['opsi b', 'pilihan b', 'b']);
                        const optC = getCol(['opsi c', 'pilihan c', 'c']);
                        const optD = getCol(['opsi d', 'pilihan d', 'd']);
                        let ans = getCol(['kunci', 'jawaban']).toUpperCase().replace(/[^ABCD]/g, '');
                        if (!['A', 'B', 'C', 'D'].includes(ans)) ans = 'A';
                        const exp = getCol(['pembahasan', 'penjelasan']);

                        if (!text || (!optA && !optB)) return null;
                        return {
                            subject: subj,
                            text: text,
                            options: { A: optA || '-', B: optB || '-', C: optC || '-', D: optD || '-' },
                            answer: ans,
                            explanation: exp
                        };
                    }).filter(Boolean);

                    renderQuestionImportPreview(parsedQuestions);
                } catch (err) {
                    showToast('Gagal membaca file Excel: ' + err.message, 'error');
                }
            };
            reader.readAsArrayBuffer(file);
        }

        function handleDocQuestionUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            const fileName = file.name.toLowerCase();
            if (fileName.endsWith('.docx') && window.mammoth) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    mammoth.extractRawText({ arrayBuffer: e.target.result })
                        .then(result => {
                            document.getElementById('docPasteTextarea').value = result.value;
                            parsePastedDocText();
                        })
                        .catch(err => showToast('Gagal mengekstrak DOCX: ' + err.message, 'error'));
                };
                reader.readAsArrayBuffer(file);
            } else {
                // Fallback pembaca .doc (HTML/RTF/Plain text)
                const reader = new FileReader();
                reader.onload = function(e) {
                    const rawText = e.target.result;
                    const tempDiv = document.createElement('div');
                    tempDiv.innerHTML = rawText;
                    const cleanText = tempDiv.textContent || tempDiv.innerText || rawText;
                    document.getElementById('docPasteTextarea').value = cleanText.trim();
                    parsePastedDocText();
                };
                reader.readAsText(file);
            }
        }

        function parsePastedDocText() {
            const rawText = document.getElementById('docPasteTextarea').value;
            const defaultSubject = document.getElementById('importDefaultSubject').value;
            const parsed = parseStructuredQuestionText(rawText, defaultSubject);
            renderQuestionImportPreview(parsed);
        }

            }

            const statusEl = document.getElementById('pdfExtractionStatus');
            statusEl.textContent = 'Sedang mengekstrak teks dari dokumen PDF...';

            try {
                const arrayBuffer = await file.arrayBuffer();
                const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
                let fullText = '';

                for (let pageNum = 1; pageNum <= pdf.numPages; pageNum++) {
                    const page = await pdf.getPage(pageNum);
                    const textContent = await page.getTextContent();
                    const pageStrings = textContent.items.map(item => item.str);
                    fullText += pageStrings.join('\n') + '\n';
                }

                statusEl.textContent = `Selesai membaca ${pdf.numPages} halaman PDF.`;
                const defaultSubject = document.getElementById('importDefaultSubject').value;
                const parsed = parseStructuredQuestionText(fullText, defaultSubject);
                renderQuestionImportPreview(parsed);
            } catch (err) {
                statusEl.textContent = 'Gagal membaca file PDF.';
                showToast('Gagal memproses file PDF: ' + err.message, 'error');
            }
        }

        // Parser Teks Soal Terstruktur (Mendukung DOC, Paste Teks, & PDF)
        function parseStructuredQuestionText(rawText, defaultSubject) {
            if (!rawText || !rawText.trim()) return [];
            const lines = rawText.split(/\r?\n/).map(l => l.trim()).filter(Boolean);
            const questions = [];
            let currentQ = null;

            const qStartRegex = /^(\d+)[\.\)]\s+(.+)/;
            const optRegex = /^([A-Da-d])[\.\)]\s*(.+)/;
            const keyRegex = /^(?:kunci|jawaban|kunci jawaban)\s*[:\-]\s*([A-Da-d])/i;
            const expRegex = /^(?:pembahasan|penjelasan)\s*[:\-]\s*(.+)/i;

            lines.forEach(line => {
                const qMatch = line.match(qStartRegex);
                if (qMatch) {
                    if (currentQ && currentQ.text && (currentQ.options.A || currentQ.options.B)) {
                        questions.push(currentQ);
                    }
                    currentQ = {
                        subject: defaultSubject,
                        text: qMatch[2].trim(),
                        options: { A: '', B: '', C: '', D: '' },
                        answer: 'A',
                        explanation: ''
                    };
                    return;
                }

                if (!currentQ) return;

                const optMatch = line.match(optRegex);
                if (optMatch) {
                    const letter = optMatch[1].toUpperCase();
                    // Cek apakah opsi A, B, C, D tertulis dalam 1 baris yang sama
                    const inlineSplit = line.split(/\s+(?=[B-Db-d][\.\)]\s+)/);
                    if (inlineSplit.length > 1) {
                        inlineSplit.forEach(part => {
                            const subMatch = part.match(optRegex);
                            if (subMatch) {
                                currentQ.options[subMatch[1].toUpperCase()] = subMatch[2].trim();
                            }
                        });
                    } else {
                        currentQ.options[letter] = optMatch[2].trim();
                    }
                    return;
                }

                const keyMatch = line.match(keyRegex);
                if (keyMatch) {
                    currentQ.answer = keyMatch[1].toUpperCase();
                    return;
                }

                const expMatch = line.match(expRegex);
                if (expMatch) {
                    currentQ.explanation = expMatch[1].trim();
                    return;
                }

                // Jika lanjutan kalimat pertanyaan sebelum opsi A diisi
                if (!currentQ.options.A && !currentQ.options.B) {
                    currentQ.text += ' ' + line;
                }
            });

            if (currentQ && currentQ.text && (currentQ.options.A || currentQ.options.B)) {
                questions.push(currentQ);
            }

            return questions;
        }

        function handleExcelStudentUpload(event) {
            const file = event.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    let rows = [];
                    if (window.XLSX) {
                        const data = new Uint8Array(e.target.result);
                        const workbook = XLSX.read(data, { type: 'array' });
                        const firstSheet = workbook.Sheets[workbook.SheetNames[0]];
                        rows = XLSX.utils.sheet_to_json(firstSheet, { defval: '' });
                    } else {
                        const text = new TextDecoder().decode(e.target.result);
                        rows = parseSimpleCSV(text);
                    }

                    const existingUsernames = new Set(appState.students.map(s => s.username.toLowerCase()));
                    const seenInFile = new Set();
                    let hasDuplicate = false;

                    const parsedStudents = rows.map((row, idx) => {
                        const getCol = (keys) => {
                            for (const k of Object.keys(row)) {
                                if (keys.some(target => k.trim().toLowerCase().includes(target))) {
                                    return String(row[k]).trim();
                                }
                            }
                            return '';
                        };

                        const name = getCol(['nama']);
                        const username = getCol(['nis', 'username', 'user']);
                        const password = getCol(['password', 'sandi', 'pass']) || '123456';
                        const cls = getCol(['kelas', 'rombel']) || 'Kelas 5';

                        if (!name || !username) return null;

                        const lowerUser = username.toLowerCase();
                        const isDuplicate = existingUsernames.has(lowerUser) || seenInFile.has(lowerUser);
                        if (isDuplicate) hasDuplicate = true;
                        seenInFile.add(lowerUser);

                        return { name, username, password, class: cls, isDuplicate };
                    }).filter(Boolean);

                    pendingImportedStudents = parsedStudents;
                    const section = document.getElementById('studentImportPreviewSection');
                    const tbody = document.getElementById('studentImportPreviewBody');
                    const countSpan = document.getElementById('previewStudentCount');
                    const dupWarn = document.getElementById('previewDuplicateWarning');
                    const saveBtn = document.getElementById('btnConfirmImportStudents');

                    if (parsedStudents.length === 0) {
                        section.classList.add('hidden');
                        saveBtn.disabled = true;
                        showToast('Tidak ditemukan data siswa yang valid pada file Excel.', 'warning');
                        return;
                    }

                    countSpan.textContent = parsedStudents.length;
                    if (hasDuplicate) dupWarn.classList.remove('hidden');
                    else dupWarn.classList.add('hidden');

                    tbody.innerHTML = parsedStudents.map((s, i) => `
                        <tr class="${s.isDuplicate ? 'bg-amber-50/70' : 'hover:bg-slate-50'}">
                            <td class="py-2 px-3 text-slate-500">${i + 1}</td>
                            <td class="py-2 px-3 font-mono font-semibold text-slate-800">${s.username}</td>
                            <td class="py-2 px-3 font-medium text-slate-800">${s.name}</td>
                            <td class="py-2 px-3">${s.class}</td>
                            <td class="py-2 px-3 font-mono text-slate-600">${s.password}</td>
                            <td class="py-2 px-3 text-center">
                                ${s.isDuplicate
                                    ? '<span class="px-2 py-0.5 rounded-full text-[10px] font-bold bg-amber-200 text-amber-900">Duplikat (Lewati)</span>'
                                    : '<span class="px-2 py-0.5 rounded-full text-[10px] font-bold bg-emerald-100 text-emerald-700">Siap Impor</span>'}
                            </td>
                        </tr>
                    `).join('');

                    section.classList.remove('hidden');
                    const validCount = parsedStudents.filter(s => !s.isDuplicate).length;
                    saveBtn.disabled = validCount === 0;
                    showToast(`Terbaca ${parsedStudents.length} baris (${validCount} siswa baru siap diimpor).`);
                } catch (err) {
                    showToast('Gagal membaca file Excel siswa: ' + err.message, 'error');
                }
            };
            reader.readAsArrayBuffer(file);
        }

        function downloadTextFile(content, filename, mimeType) {
            const blob = new Blob([content], { type: mimeType });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');

        function parseSimpleCSV(csvText) {
            const lines = csvText.split(/\r?\n/).filter(l => l.trim());
            if (lines.length < 2) return [];
            const headers = lines[0].split(',').map(h => h.replace(/^"|"$/g, '').trim());
            return lines.slice(1).map(line => {
                const cols = line.split(',').map(c => c.replace(/^"|"$/g, '').trim());
                const obj = {};
                headers.forEach((h, i) => { obj[h] = cols[i] || ''; });
                return obj;
            });
        }

            }
            return token;
        }

        function startExamSession(sessionId) {
            const ses = appState.sessions.find(s => s.id === sessionId);
            if (!ses) return;

            const enteredToken = document.getElementById(`tokenInput_${sessionId}`).value.trim().toUpperCase();
            if (enteredToken !== ses.token.toUpperCase()) {
                showModal('Token Masuk Salah', `Token yang Anda masukkan tidak sesuai. Silakan minta Token Masuk kepada Guru / Pengawas Ruang (Petunjuk demo: <strong>${ses.token}</strong>).`);
                return;
            }

            let questionsPool = ses.subject === 'ALL'
                ? [...appState.questions]
                : appState.questions.filter(q => q.subject === ses.subject);

            if (questionsPool.length === 0) {
                showModal('Soal Belum Tersedia', 'Belum ada butir soal pada mata pelajaran sesi ujian ini.');
                return;
            }

            if (ses.randomize) {
                questionsPool.sort(() => Math.random() - 0.5);
            }

            activeExam = {
                session: ses,
                questions: questionsPool,
                currentIndex: 0,
                answers: {}, // { questionId: 'A' | 'B' | 'C' | 'D' }
                flags: {},   // { questionId: true }
                remainingSeconds: ses.duration * 60
            };

            document.getElementById('examWorkspaceTitle').textContent = ses.title;
            document.getElementById('examWorkspaceStudentInfo').textContent = `${currentUser.data.name} (${currentUser.data.class})`;

            navigateTo('exam-workspace');
            renderExamQuestion();
            startExamCountdown();
        }

        function startExamCountdown() {
            if (examTimerInterval) clearInterval(examTimerInterval);
            updateTimerDisplay();

            examTimerInterval = setInterval(() => {
                if (!activeExam) {
                    clearInterval(examTimerInterval);
                    return;
                }
                activeExam.remainingSeconds--;
                updateTimerDisplay();

                if (activeExam.remainingSeconds <= 0) {
                    clearInterval(examTimerInterval);
                    showToast('Waktu ujian telah habis! Jawaban dikumpulkan otomatis.', 'warning');
                    finishAndGradeExam();
                }
            }, 1000);
        }

        function updateTimerDisplay() {
            const mins = Math.floor(activeExam.remainingSeconds / 60);
            const secs = activeExam.remainingSeconds % 60;
            const display = `${String(mins).padStart(2, '0')}:${String(secs).padStart(2, '0')}`;
            const el = document.getElementById('examTimerDisplay');
            el.textContent = display;

            if (activeExam.remainingSeconds <= 60) {
                el.classList.add('text-rose-300');
            } else {
                el.classList.remove('text-rose-300');
            }
        }

        function renderExamQuestion() {
            if (!activeExam) return;
            const idx = activeExam.currentIndex;
            const q = activeExam.questions[idx];
            const total = activeExam.questions.length;

            document.getElementById('currentQuestionNumberBadge').textContent = `${idx + 1}`;
            document.getElementById('currentQuestionSubjectBadge').textContent = q.subject;
            document.getElementById('examQuestionText').textContent = q.text;

            // Render Pilihan Ganda
            const selectedAns = activeExam.answers[q.id];
            const optionsContainer = document.getElementById('examOptionsContainer');
            optionsContainer.innerHTML = ['A', 'B', 'C', 'D'].map(optKey => {
                const isSelected = selectedAns === optKey;
                return `
                    <div onclick="selectExamOption('${q.id}', '${optKey}')"
                        class="p-3.5 sm:p-4 rounded-xl border-2 transition cursor-pointer flex items-center gap-3.5 select-none ${isSelected ? 'border-[#0099FF] bg-sky-50/80 shadow-xs' : 'border-slate-200 hover:border-sky-300 hover:bg-slate-50'}">
                        <div class="w-8 h-8 rounded-full flex items-center justify-center font-bold text-sm shrink-0 transition ${isSelected ? 'bg-[#0099FF] text-white' : 'bg-slate-100 text-slate-700 border border-slate-300'}">
                            ${optKey}
                        </div>
                        <div class="text-sm sm:text-base text-slate-800 ${isSelected ? 'font-semibold' : ''}">
                            ${q.options[optKey]}
                        </div>
                    </div>
                `;
            }).join('');

            // Update tombol Prev/Next/Flag
            document.getElementById('btnPrevQuestion').disabled = idx === 0;
            const nextBtn = document.getElementById('btnNextQuestion');
            if (idx === total - 1) {
                nextBtn.innerHTML = `<span>Selesai</span> <i class="fas fa-check"></i>`;
                nextBtn.onclick = confirmFinishExam;
                nextBtn.classList.replace('bg-[#0099FF]', 'bg-emerald-600');
            } else {
                nextBtn.innerHTML = `<span>Selanjutnya</span> <i class="fas fa-chevron-right"></i>`;
                nextBtn.onclick = nextQuestion;
                nextBtn.classList.replace('bg-emerald-600', 'bg-[#0099FF]');
            }

            const isFlagged = !!activeExam.flags[q.id];
            document.getElementById('btnFlagText').textContent = isFlagged ? 'Hapus Ragu-ragu' : 'Ragu-ragu';

            renderExamNavigationGrid();
        }

        function selectExamOption(questionId, optionKey) {
            if (!activeExam) return;
            activeExam.answers[questionId] = optionKey;
            renderExamQuestion();
        }

        function toggleFlagQuestion() {
            if (!activeExam) return;
            const q = activeExam.questions[activeExam.currentIndex];
            activeExam.flags[q.id] = !activeExam.flags[q.id];
            renderExamQuestion();
        }

        function prevQuestion() {
            if (activeExam && activeExam.currentIndex > 0) {
                activeExam.currentIndex--;
                renderExamQuestion();
            }
        }

        function nextQuestion() {
            if (activeExam && activeExam.currentIndex < activeExam.questions.length - 1) {
                activeExam.currentIndex++;
                renderExamQuestion();
            }
        }

        function jumpToQuestion(index) {
            if (activeExam && index >= 0 && index < activeExam.questions.length) {
                activeExam.currentIndex = index;
                renderExamQuestion();
            }
        }

        function renderExamNavigationGrid() {
            const grid = document.getElementById('questionNavGrid');
            const total = activeExam.questions.length;
            const answeredCount = Object.keys(activeExam.answers).length;

            document.getElementById('answeredProgressText').textContent = `${answeredCount} / ${total} Dijawab`;

            grid.innerHTML = activeExam.questions.map((q, idx) => {
                const isCurrent = idx === activeExam.currentIndex;
                const isFlagged = !!activeExam.flags[q.id];
                const isAnswered = !!activeExam.answers[q.id];

                let colorClass = 'bg-white text-slate-700 border-slate-300 hover:bg-slate-50';
                if (isFlagged) {
                    colorClass = 'bg-amber-400 text-slate-900 border-amber-500 font-bold';
                } else if (isAnswered) {
                    colorClass = 'bg-emerald-500 text-white border-emerald-600 font-bold';
                }

                const activeRing = isCurrent ? 'ring-2 ring-offset-1 ring-[#0099FF]' : '';

                return `
                    <button onclick="jumpToQuestion(${idx})"
                        class="h-10 rounded-lg border text-xs sm:text-sm flex items-center justify-center transition cursor-pointer ${colorClass} ${activeRing}">
                        ${idx + 1}
                    </button>
                `;
            }).join('');
        }

        function confirmFinishExam() {
            const total = activeExam.questions.length;
            const answeredCount = Object.keys(activeExam.answers).length;
            const unanswered = total - answeredCount;
            const flaggedCount = Object.values(activeExam.flags).filter(Boolean).length;

            let warningText = 'Apakah Anda yakin ingin mengakhiri ujian dan mengumpulkan jawaban sekarang?';
            if (unanswered > 0 || flaggedCount > 0) {
                warningText += `<br><br><span class="text-amber-700 font-semibold">Perhatian:</span> Masih ada <strong>${unanswered} soal belum dijawab</strong> dan <strong>${flaggedCount} soal ragu-ragu</strong>.`;
            }

            showModal(
                'Konfirmasi Selesai Ujian',
                warningText,
                [
                    { text: 'Kembali Mengerjakan', class: 'bg-slate-200 text-slate-700', onClick: closeModal },
                    {
                        text: 'Ya, Kumpulkan Jawaban',
                        class: 'bg-emerald-600 text-white hover:bg-emerald-700',
                        onClick: () => {
                            closeModal();
                            finishAndGradeExam();
                        }
                    }
                ]
            );
        }

        function finishAndGradeExam() {
            if (!activeExam) return;
            if (examTimerInterval) clearInterval(examTimerInterval);

            const { session, questions, answers } = activeExam;
            let correctCount = 0;

            questions.forEach(q => {
                if (answers[q.id] === q.answer) {
                    correctCount++;
                }
            });

            const total = questions.length;
            const score = Math.round((correctCount / total) * 100);
            const passed = score >= session.kkm;

            const now = new Date();
            const timestamp = `${String(now.getDate()).padStart(2, '0')}/${String(now.getMonth() + 1).padStart(2, '0')}/${now.getFullYear()} ${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;

            const resultRecord = {
                id: 'res_' + Date.now(),
                studentId: currentUser.data.id,
                studentName: currentUser.data.name,
                studentClass: currentUser.data.class,
                sessionId: session.id,
                sessionTitle: session.title,
                correct: correctCount,
                total: total,
                score: score,
                kkm: session.kkm,
                passed: passed,
                timestamp: timestamp
            };

            appState.results.push(resultRecord);
            saveAppState();

            // Tampilkan layar hasil ujian
            document.getElementById('resultSessionTitle').textContent = session.title;
            document.getElementById('resultCorrectCount').textContent = `${correctCount} / ${total}`;
            document.getElementById('resultFinalScore').textContent = score;
            document.getElementById('resultKKMVal').textContent = session.kkm;

            const badge = document.getElementById('resultStatusBadge');
            if (passed) {
                badge.textContent = 'TUNTAS / LULUS';
                badge.className = 'px-3 py-1 rounded-full text-xs font-bold uppercase bg-emerald-100 text-emerald-700';
            } else {
                badge.textContent = 'BELUM TUNTAS';
                badge.className = 'px-3 py-1 rounded-full text-xs font-bold uppercase bg-rose-100 text-rose-700';
            }

            // Pembahasan soal
            const reviewSection = document.getElementById('resultReviewSection');
            const reviewList = document.getElementById('resultReviewList');
            if (session.showReview) {
                reviewSection.classList.remove('hidden');
                reviewList.innerHTML = questions.map((q, idx) => {
                    const userAns = answers[q.id] || '-';
                    const isRight = userAns === q.answer;
                    return `
                        <div class="p-4 rounded-xl border ${isRight ? 'border-emerald-200 bg-emerald-50/40' : 'border-rose-200 bg-rose-50/40'} text-xs sm:text-sm space-y-1.5">
                            <div class="flex items-center justify-between">
                                <span class="font-bold text-slate-700">Soal #${idx + 1} (${q.subject})</span>
                                <span class="font-bold ${isRight ? 'text-emerald-600' : 'text-rose-600'}">
                                    ${isRight ? '<i class="fas fa-check-circle"></i> Benar' : '<i class="fas fa-times-circle"></i> Kurang Tepat'}
                                </span>
                            </div>
                            <p class="text-slate-800">${q.text}</p>
                            <div class="pt-1 flex flex-wrap gap-4 text-xs">
                                <span>Jawaban Anda: <strong>${userAns}</strong></span>
                                <span class="text-emerald-700 font-semibold">Kunci Jawaban: <strong>${q.answer}. ${q.options[q.answer]}</strong></span>
                            </div>
                        </div>
                    `;
                }).join('');
            } else {
                reviewSection.classList.add('hidden');
            }

            activeExam = null;
            navigateTo('exam-result');
        }
